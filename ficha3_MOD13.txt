CREATE TABLE `fornecimento` (
    `CodFornecedor` INT NOT NULL,
    `CodProduto` INT NOT NULL,
    `Data_Fornecimento` DATE DEFAULT NULL,
    PRIMARY KEY (`CodFornecedor`, `CodProduto`),
    CONSTRAINT `fk_fornecimento_fornecedor`
        FOREIGN KEY (`CodFornecedor`)
        REFERENCES `fornecedores` (`CodFornecedor`),
    CONSTRAINT `fk_fornecimento_produto`
        FOREIGN KEY (`CodProduto`)
        REFERENCES `produtos` (`CodProduto`)
        'Ficha N3 PSI - GITHUB'
) ENGINE=InnoDB;
